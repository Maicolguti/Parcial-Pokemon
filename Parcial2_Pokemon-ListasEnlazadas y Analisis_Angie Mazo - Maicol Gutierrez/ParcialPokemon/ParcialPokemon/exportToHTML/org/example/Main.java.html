package org.example;

import org.example.Model.Pokemon;
import org.example.Model.LineaEvolutiva;
import org.example.Service.EntrenamientoMasivo;
import java.util.Random;

public class Main {
    public static void main(String[] args) {

        // ============================================
        // 1. CREAR LA CADENA EVOLUTIVA
        // Se crean los 3 nodos y se enlazan entre sí.
        // ============================================
        Pokemon charizard = new Pokemon("Charizard", 78, 84, 78, -1);
        Pokemon charmeleon = new Pokemon("Charmeleon", 58, 64, 58, 5000);
        Pokemon charmander = new Pokemon("Charmander", 39, 52, 43, 1500);

        charmander.setEvolucionSiguiente(charmeleon);
        charmeleon.setEvolucionSiguiente(charizard);

        // ============================================
        // 2. PRUEBA UNITARIA: Charmander vs Rattata
        // ============================================
        LineaEvolutiva miPokemonPrueba = new LineaEvolutiva(charmander);
        Pokemon rattata = new Pokemon("Rattata", 30, 56, 35, -1);

        int hpRattata = rattata.getPuntosDeVidaMaximos();
        int hpCharmander = miPokemonPrueba.getFaseActual().getPuntosDeVidaMaximos();

        System.out.println("=== Prueba unitaria: Charmander vs Rattata ===");

        int danio1 = Math.max(1, miPokemonPrueba.getFaseActual().getAtaque() - rattata.getDefensa());
        hpRattata -= danio1;
        System.out.println("Turno 1 - Charmander ataca. Daño: " + danio1 + " | Rattata HP: " + hpRattata);

        if (hpRattata > 0) {
            int danio2 = Math.max(1, rattata.getAtaque() - miPokemonPrueba.getFaseActual().getDefensa());
            hpCharmander -= danio2;
            System.out.println("Turno 2 - Rattata ataca. Daño: " + danio2 + " | Charmander HP: " + hpCharmander);
        }

        if (hpRattata > 0) {
            int danio3 = Math.max(1, miPokemonPrueba.getFaseActual().getAtaque() - rattata.getDefensa());
            hpRattata -= danio3;
            System.out.println("Turno 3 - Charmander ataca. Daño: " + danio3 + " | Rattata HP: " + hpRattata);
        }

        if (hpRattata <= 0) {
            System.out.println("¡Charmander ganó la batalla!");
            miPokemonPrueba.ganarExperiencia(50);
            miPokemonPrueba.mecanismoEvolucion();
        }

        System.out.println("XP acumulada: " + miPokemonPrueba.getExperienciaAcumulada());
        System.out.println("Fase actual: " + miPokemonPrueba.getFaseActual().getNombre());

        // ============================================
        // 3. PRUEBA DE ESTRÉS: horda de 100,000 Caterpie generados aleatoriamente
        // ============================================
        System.out.println("\n=== Prueba de estrés: entrenamiento masivo ===");

        // LineaEvolutiva NUEVA para esta prueba, con XP en 0
        LineaEvolutiva miPokemonEntrenamiento = new LineaEvolutiva(charmander);

        // Random genera los números aleatorios para cada Caterpie
        Random random = new Random();
        Pokemon[] hordaEnemigos = new Pokemon[100000];

        for (int i = 0; i < hordaEnemigos.length; i++) {
            // random.nextInt(21) da un número entre 0 y 20; sumando el mínimo
            // desplazamos ese rango a donde queremos (ej: 30 a 50)
            int hpAleatorio = 30 + random.nextInt(21);       // entre 30 y 50
            int ataqueAleatorio = 20 + random.nextInt(16);   // entre 20 y 35
            int defensaAleatoria = 20 + random.nextInt(21);  // entre 20 y 40

            hordaEnemigos[i] = new Pokemon("Caterpie", hpAleatorio, ataqueAleatorio, defensaAleatoria, -1);
        }

        // Ejecutamos el entrenamiento masivo
        EntrenamientoMasivo entrenamiento = new EntrenamientoMasivo();
        entrenamiento.iniciarEntrenamientoMasivo(miPokemonEntrenamiento, hordaEnemigos);

        // Resultado final, después de las 100,000 batallas
        System.out.println("Fase final tras el entrenamiento: " + miPokemonEntrenamiento.getFaseActual().getNombre());
        System.out.println("XP final acumulada: " + miPokemonEntrenamiento.getExperienciaAcumulada());
    }
}
